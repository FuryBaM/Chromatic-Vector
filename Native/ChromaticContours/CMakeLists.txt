cmake_minimum_required(VERSION 3.16)
project(chromatic_contours LANGUAGES CXX)
add_library(chromatic_contours SHARED src/chromatic_contours.cpp)
target_include_directories(chromatic_contours PUBLIC include)
target_compile_features(chromatic_contours PUBLIC cxx_std_17)
set_target_properties(chromatic_contours PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(chromatic_contours PRIVATE /W4)
else()
  target_compile_options(chromatic_contours PRIVATE -Wall -Wextra -Wpedantic)
endif()
